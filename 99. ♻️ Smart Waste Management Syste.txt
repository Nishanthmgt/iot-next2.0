✅ Arduino / ESP32 Code (Advanced PID Control)
#include <PID_v1.h>
#include <Wire.h>
#include <Adafruit_SSD1306.h>

#define TRIG 5
#define ECHO 18
#define RELAY 26
#define POT A0

double Setpoint, Input, Output;
double Kp = 2.0, Ki = 5.0, Kd = 1.0;

PID myPID(&Input, &Output, &Setpoint, Kp, Ki, Kd, DIRECT);

Adafruit_SSD1306 display(128, 64, &Wire, -1);

long getDistance() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  return pulseIn(ECHO, HIGH) * 0.034 / 2;
}

void setup() {
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
  pinMode(RELAY, OUTPUT);

  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  display.clearDisplay();

  Setpoint = 20;  // desired level in cm
  myPID.SetMode(AUTOMATIC);
}

void loop() {
  Input = getDistance();
  myPID.Compute();

  if (Input > Setpoint)
    digitalWrite(RELAY, HIGH);
  else
    digitalWrite(RELAY, LOW);

  display.clearDisplay();
  display.setCursor(0, 0);
  display.print("Level: ");
  display.println(Input);

  display.setCursor(0, 20);
  display.print("Set: ");
  display.println(Setpoint);

  display.setCursor(0, 40);
  display.print("Output: ");
  display.println(Output);

  display.display();
  delay(1000);
}

✅ Arduino Code (Advanced & Functional)
#include <TinyGPS++.h>
#include <Wire.h>
#include <Adafruit_SSD1306.h>

#define GPS_RX 16
#define GPS_TX 17
#define LTE_RX 27
#define LTE_TX 26
#define PANIC_BTN 14
#define BUZZER 25

TinyGPSPlus gps;
HardwareSerial gpsSerial(1);
HardwareSerial lteSerial(2);

Adafruit_SSD1306 display(128, 64, &Wire, -1);

void setup() {
  Serial.begin(115200);

  pinMode(PANIC_BTN, INPUT_PULLUP);
  pinMode(BUZZER, OUTPUT);

  gpsSerial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
  lteSerial.begin(115200, SERIAL_8N1, LTE_RX, LTE_TX);

  Wire.begin(21, 22);
  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);

  display.clearDisplay();
  display.setCursor(10, 20);
  display.println("Elderly Care System");
  display.display();
  delay(2000);
}

void loop() {
  while (gpsSerial.available()) {
    gps.encode(gpsSerial.read());
  }

  if (digitalRead(PANIC_BTN) == LOW) {
    digitalWrite(BUZZER, HIGH);

    float lat = gps.location.lat();
    float lon = gps.location.lng();

    sendAlert(lat, lon);

    delay(5000);
    digitalWrite(BUZZER, LOW);
  }
}

void sendAlert(float lat, float lon) {
  lteSerial.println("AT+CMGF=1");
  delay(1000);
  lteSerial.println("AT+CMGS=\"+91XXXXXXXXXX\"");
  delay(1000);

  lteSerial.print("Emergency Alert!\nLocation:\n");
  lteSerial.print("https://maps.google.com/?q=");
  lteSerial.print(lat, 6);
  lteSerial.print(",");
  lteSerial.print(lon, 6);
  lteSerial.write(26);
}

#define RELAY 26
#define BUZZER 27

#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

void setup() {
  Serial.begin(115200);

  pinMode(RELAY, OUTPUT);
  pinMode(BUZZER, OUTPUT);

  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);

  display.setTextColor(WHITE);
  display.setCursor(10, 20);
  display.println("Agri-Tech Gateway");

void loop() {

  float hum = dht.readHumidity();
  int soil = analogRead(SOIL_PIN);
  int light = analogRead(LDR_PIN);

    digitalWrite(RELAY, HIGH);
    digitalWrite(BUZZER, HIGH);

  } else {
    digitalWrite(RELAY, LOW);
    digitalWrite(BUZZER, LOW);

✅ Arduino Code (Advanced & Working)
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <MAX30100_PulseOximeter.h>
#include <OneWire.h>
#include <DallasTemperature.h>

#define TEMP_PIN 4
#define BUZZER 26

PulseOximeter pox;
OneWire oneWire(TEMP_PIN);
DallasTemperature sensors(&oneWire);

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

void setup() {
  Serial.begin(115200);

  pinMode(BUZZER, OUTPUT);

  Wire.begin(21, 22);
  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);

  pox.begin();
  sensors.begin();

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(WHITE);
  display.setCursor(10, 20);
  display.println("Health Monitor");
  display.display();
  delay(2000);
}

void loop() {
  pox.update();
  sensors.requestTemperatures();

  float heartRate = pox.getHeartRate();
  float spo2 = pox.getSpO2();
  float temp = sensors.getTempCByIndex(0);

  display.clearDisplay();
  display.setCursor(0, 0);
  display.print("HR: ");
  display.print(heartRate);
  display.println(" bpm");

  display.setCursor(0, 15);
  display.print("SpO2: ");
  display.print(spo2);
  display.println(" %");

  display.setCursor(0, 30);
  display.print("Temp: ");
  display.print(temp);
  display.println(" C");

  if (heartRate > 120 || spo2 < 90 || temp > 38) {
    digitalWrite(BUZZER, HIGH);
    display.setCursor(0, 50);
    display.print("ALERT!");
  } else {
    digitalWrite(BUZZER, LOW);
    display.setCursor(0, 50);
    display.print("Normal");
  }

  display.display();
  delay(2000);
}

HardwareSerial sdsSerial(2);

  Serial.begin(115200);

#define BUZZER 26

int distance;

  Serial.begin(115200);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(BUZZER, OUTPUT);

🧠 Arduino Code (ESP32 + GPS + GSM)
#include <TinyGPS++.h>
#include <HardwareSerial.h>

TinyGPSPlus gps;
HardwareSerial gpsSerial(1);
HardwareSerial gsmSerial(2);

#define GPS_RX 4
#define GPS_TX 5
#define GSM_RX 16
#define GSM_TX 17

void setup() {
  Serial.begin(115200);

  gpsSerial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
  gsmSerial.begin(9600, SERIAL_8N1, GSM_RX, GSM_TX);

  Serial.println("Smart Vehicle Tracking Started");
}

void loop() {
  while (gpsSerial.available()) {
    gps.encode(gpsSerial.read());
  }

  if (gps.location.isUpdated()) {
    float lat = gps.location.lat();
    float lon = gps.location.lng();

    Serial.print("Lat: "); Serial.println(lat);
    Serial.print("Lon: "); Serial.println(lon);

    sendSMS(lat, lon);
    delay(30000);
  }
}

void sendSMS(float lat, float lon) {
  gsmSerial.println("AT+CMGF=1");
  delay(1000);
  gsmSerial.println("AT+CMGS=\"+91XXXXXXXXXX\"");
  delay(1000);

  gsmSerial.print("Vehicle Location:\n");
  gsmSerial.print("https://maps.google.com/?q=");
  gsmSerial.print(lat, 6);
  gsmSerial.print(",");
  gsmSerial.print(lon, 6);
  gsmSerial.write(26);
}
